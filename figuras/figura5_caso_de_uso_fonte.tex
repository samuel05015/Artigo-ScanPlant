\newcommand{\actor}[3]{%
  \coordinate (#1) at (#2,#3);
  \draw[line width=0.72pt] (#2,#3+0.42) circle (0.18);
  \draw[line width=0.72pt] (#2,#3+0.24) -- (#2,#3-0.25);
  \draw[line width=0.72pt] (#2-0.30,#3+0.05) -- (#2,#3+0.13) -- (#2+0.30,#3+0.05);
  \draw[line width=0.72pt] (#2,#3-0.25) -- (#2-0.27,#3-0.62);
  \draw[line width=0.72pt] (#2,#3-0.25) -- (#2+0.27,#3-0.62);
}
